    setState(() => _presentation = value.copyWith({'customized':true}));
  }

    alignment:Alignment.centerRight,

      icon:const Icon(Icons.my_location,size:18),label:const Text('定位預覽')),

  Future<void> _locatePreview(String section) async {
    final shown = switch(section) {
      'features' => environmentFeatures.isNotEmpty,
      'gallery' => environmentGalleryImages.isNotEmpty,
      _ => environmentFacilityOptions.any((item)=>selectedFacilityKeys.contains(item['key']) &&
        !_presentation.hiddenFacilityKeys.contains(item['key'])) || _presentation.services.any((s)=>s.visible),
    };
    if(!shown){
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content:Text('此區塊目前沒有顯示內容')));
      return;
    }
    if(!_inlinePreviewVisible || _previewKey.currentState==null){
      await _openDraftPreview(locate:section);
      return;
    }
    await _previewKey.currentState!.reveal(_previewSections[section]!);
  }

  void _retireServicePhoto(EnvironmentServiceCard service) {
    if(service.imageUrl.isEmpty && service.imagePath.isEmpty)return;
    _trackRetiredOfficial(slot:'service',path:service.imagePath,url:service.imageUrl);
  }

  void _removeServiceImage(String id) {
    if(_imageBusy)return;
    final services=_presentation.services;
    final index=services.indexWhere((s)=>s.id==id);
    if(index<0)return;
    _retireServicePhoto(services[index]);
    services[index]=services[index].copyWith({'imageUrl':'','imageStoragePath':''});
    _setPresentation(_presentation.copyWith({'customServices':services.map((s)=>s.toMap()).toList()}));
  }

  Future<void> _deleteService(String id) async {
    if(_imageBusy)return;
    final confirmed=await showDialog<bool>(context:context,builder:(context)=>AlertDialog(
      title:const Text('刪除自訂服務？'),content:const Text('按儲存後才會從正式內容移除。'),
      actions:[TextButton(onPressed:()=>Navigator.pop(context,false),child:const Text('取消')),
        FilledButton(onPressed:()=>Navigator.pop(context,true),child:const Text('刪除'))]));
    if(confirmed!=true||!mounted)return;
    final services=_presentation.services;
    final index=services.indexWhere((s)=>s.id==id);
    if(index<0)return;
    _retireServicePhoto(services.removeAt(index));
    _setPresentation(_presentation.copyWith({'customServices':services.map((s)=>s.toMap()).toList()}));
  }

  Future<void> _uploadServiceImage(String id) async {
    if(_imageBusy)return;
    setState((){_pickingImage=true;_busyMessage='選擇服務照片…';});
    try {
      final image=await ImagePicker().pickImage(source:ImageSource.gallery);
      if(image==null||!mounted)return;
      final bytes=await image.readAsBytes();
      if(!mounted)return;
      if(bytes.isEmpty||bytes.length>_maxImageBytes){_showImageError('單張圖片須為有效圖片，且不超過 5 MB');return;}
      final cropped=await FixedAspectImageCropPage.open(context:context,imageBytes:bytes,
        spec:FixedImageSpec.environmentFeature,title:'裁切服務照片');
      if(cropped==null||!mounted)return;
      if(cropped.isEmpty||cropped.length>_maxImageBytes){_showImageError('處理後的圖片不可超過 5 MB');return;}
      setState(()=>_busyMessage='正在上傳服務照片…');
      final uploaded=await ShopService.instance.uploadEnvironmentFeatureImage(shopId:widget.shopId,bytes:cropped);
      final tracked=_TrackedEnvImage(slot:'service',path:uploaded.imageStoragePath,url:uploaded.imageUrl);
      if(!mounted){await _cleanupTracked(tracked,reason:'closed-service-upload');return;}
      _pendingUploads.add(tracked);
      final services=_presentation.services;
      final index=services.indexWhere((s)=>s.id==id);
      if(index<0)return;
      _retireServicePhoto(services[index]);
      services[index]=services[index].copyWith({'imageUrl':uploaded.imageUrl,'imageStoragePath':uploaded.imageStoragePath});
      _setPresentation(_presentation.copyWith({'customServices':services.map((s)=>s.toMap()).toList()}));
    } catch(_){
      _showImageError('服務照片處理失敗，請稍後再試');
    } finally {
      if(mounted)setState((){_pickingImage=false;_busyMessage=null;});
    }
  }
